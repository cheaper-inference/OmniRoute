## បញ្ជីសារពើភណ្ឌ Gate និងទម្រង់ប្រតិបត្តិ

### ការទទួលបេក្ខជន

លំហូរការងារ CI និង Quality Gates នីមួយៗបញ្ចេញសេចក្តីសម្រេចដែលមានស្ថិរភាព៖ `Gate / CI` និង
`Gate / Quality`។ គោលការណ៍ទទួលដែលមានកំណែរបស់វា រាយបញ្ជីរាល់ការងារខាងលើថា ជាការងារដែលត្រូវការ ឬជាការងារផ្តល់យោបល់។ ការងារដែលត្រូវការនិងអនុវត្តបាន ត្រូវតែជោគជ័យ៖ លទ្ធផលដែលបាត់ លុបចោល រំលង កំពុងរង់ចាំ និងមិនស្គាល់ មិនអាចបង្កើត PASS បានទេ។ ការចាត់ថ្នាក់ docs-only ឬ catalog-only ដែលមានសុពលភាព អាចធ្វើឱ្យ code lane មិនអនុវត្តបាន; draft PR មិនមែនជាបេក្ខជនដែលទទួលយកទេ។ ស្លាក `hotfix` មិនលើកលែងតម្រូវការភស្តុតាងឡើយ។

លំហូរការងារទាំងពីរនេះគ្របដណ្តប់ PRs ទៅកាន់សាខា main/release, ការរុញទៅ `main`, ការបញ្ជូនដោយដៃ
និងព្រឹត្តិការណ៍ merge-group។ Push, dispatch និង merge-group ដំណើរការការជ្រើសរើសពេញលេញ។ ការរុញទៅ `release/**` មិនចាប់ផ្តើមវាទេ (2026-10-18)៖ រាល់ការផ្លាស់ប្តូរបានដំណើរការម៉ាទ្រីសរួចហើយនៅលើ PR របស់វា
ហើយការរួមបញ្ចូលដែលបានបញ្ចូលគ្នា គឺជាការទទួលខុសត្រូវរបស់ Release-Green (`docs/ops/RUNNER_BOX.md` → Hosted lane)។
Forks និង merge groups ប្រើ hosted runners សម្រាប់ការងារដែលជ្រើសរើស self-hosted
runners ក្នុងករណីធម្មតា; ត្រូវផ្ទៀងផ្ទាត់សមត្ថភាព hosted ដែលគ្រប់គ្រាន់ មុនពេលដាក់ឱ្យដំណើរការ។

បង្កាន់ដៃ JSON នីមួយៗកំណត់អត្តសញ្ញាណ SHA ដែលបាន checkout, workflow run និង attempt។
CLI បដិសេធនៅពេល SHA របស់ checkout/event មិនត្រូវគ្នា។ តេស្ត workflow ភ្ជាប់សមាជិកភាពគោលការណ៍ទៅនឹងបញ្ជី `needs` របស់ការងារ verdict ដើម្បីឱ្យ lane ដែលបានបន្ថែម ឬដកចេញ មិនអាចបាត់ខ្លួនដោយស្ងាត់បាន។ បង្កាន់ដៃទាំងនេះគ្របដណ្តប់តែ workflow របស់ខ្លួនប៉ុណ្ណោះ មិនមែនការបោះពុម្ពផ្សាយ ការដាក់ឱ្យប្រើប្រាស់ ឬផ្នែកខាងក្នុងនៃ advisory scanner ដែលមានស្រាប់ទេ។ ការធ្វើឱ្យឈ្មោះត្រួតពិនិត្យទាំងពីរដំណើរការនៅក្នុងច្បាប់សាខា គឺជាការផ្លាស់ប្តូរផ្នែករដ្ឋបាលដាច់ដោយឡែក; ការបន្ថែមការងារទាំងនេះដោយខ្លួនវា មិនការពារសាខាឡើយ។

### បញ្ជីសារពើភណ្ឌការស្កេនឋិតិវន្ត

សារពើភណ្ឌ npm-alias ដែលមានកំណែ និងសមាជិកភាព static-scan ស្ថិតនៅក្នុង
`config/quality/gate-manifest.json`។ ដំណើរការ `npm run check:gate-manifest` ដើម្បីផ្ទៀងផ្ទាត់ឈ្មោះស្គ្រីប និងពាក្យបញ្ជាជាក់លាក់ប្រឆាំងនឹង `package.json`; ការបន្ថែម ការដកចេញ និងការប្រែប្រួលពាក្យបញ្ជា នឹងធ្វើឱ្យ local hook និងការងារចាត់ថ្នាក់ការផ្លាស់ប្តូរនៅក្នុង CI បរាជ័យទាំងពីរ។ Alias មួយមិនមែនជា workflow job, matrix instance ឬ test case ទេ៖ ចំនួនទាំងនេះមិនត្រូវបង្ហាញថាអាចផ្លាស់ប្តូរជំនួសគ្នាបានឡើយ។

ប្រើ `npm run quality:scan -- --list` ឬ `npm run quality:scan:fast -- --list`
ដើម្បីពិនិត្យ alias ដែលបានជ្រើស ដោយមិនប្រតិបត្តិវា។ Runner ហៅ npm entrypoint ដូច្នេះ runtime របស់វា (រួមទាំង Bun នៅពេលបានកំណត់) ត្រូវបានរក្សាទុក។ Manifest កត់ត្រា alias នៅក្រៅ profile ទាំងនោះថា ត្រូវបានហៅដាច់ដោយឡែក ហើយ maintenance commands ត្រូវបានហាមឃាត់នៅក្នុង read-only scan profiles។

Profile ទាំងនេះគ្របដណ្តប់តែ static scan ប៉ុណ្ណោះ។ វាមិនបញ្ជាក់ពី product tests, coverage, packaging, external checks ឬការទទួលយក release ពេញលេញរបស់បេក្ខជនឡើយ។ ការទទួល workflow ប្រើ `config/quality/admission-policy.json` និង `scripts/quality/admission-verdict.mjs` ដែលបានភ្ជាប់។ Release-observer profiles នៅដាច់ដោយឡែក; សូមពិនិត្យ checks និងបង្កាន់ដៃដែលអនុវត្តបានរបស់វាដោយឯករាជ្យ។ បញ្ជីសារពើភណ្ឌជាអត្ថបទខាងក្រោម គឺជាឯកសារយោង មិនមែនជាភស្តុតាងថា gate មួយពិតជាបានដំណើរការទេ។

ស្គ្រីបស្ថិតនៅក្រោម `scripts/check/` (policy gates) និង `scripts/quality/` (ratchet engine)។
ប្រភពការពិតសម្រាប់ CI គឺ `.github/workflows/ci.yml`។

### ផ្លូវលឿនសម្រាប់ Release PR (`quality.yml`)

`.github/workflows/quality.yml` បំពេញបន្ថែម CI នៅលើ main/release PRs, ការរុញទៅ `main`,
dispatch និង merge groups។ PRs ប្រើ fast checks ដែលបានច្រោះតាម path។ duplicate build ដែលបានបិទជាអចិន្ត្រៃយ៍ ត្រូវបានដកចេញ; checks ពិតប្រាកដសម្រាប់ build/package/boot នៅតែមាននៅក្នុង CI។

| Job                                              | Scope                                                                                                                                                                                                      | Blocking             |
| ------------------------------------------------ | ---------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- | -------------------- |
| `Docs Gates (fast-path)`                         | PRs ឯកសារ/កូដ; សេចក្តីយោង API docs និង docs-all                                                                                                                                                            | Yes                  |
| `Fast Quality Gates`                             | PRs កូដ; static checks, typecheck, dashboard typecheck, impacted unit tests                                                                                                                                | Yes                  |
| `Forgotten sibling tests`                        | PRs កូដ; ម៉ូឌុលដែលបានផ្លាស់ប្តូរ តាមដានទៅកាន់ static consumers និង candidate sibling tests; barrel និង dynamic-import paths ត្រូវបានរាយការណ៍ជារោគវិនិច្ឆ័យផ្តល់យោបល់ ដោយមាន allowlist exceptions ដែលបានយោង | **Advisory**         |
| `Vitest (fast-path)`                             | PRs កូដ; fast vitest suite                                                                                                                                                                                 | Yes                  |

ចាប់បានការត្រឡប់ថយក្រោយផ្នែក i18n មួយដែល gate ផ្សេងទៀតមិនអាចមើលឃើញតាមរចនាសម្ព័ន្ធបាន៖ តម្លៃភាសាអង់គ្លេសមួយត្រូវបានសរសេរឡើងវិញ ហើយការបកប្រែដែលបានបង្កើតពីភាសាអង់គ្លេស _មុន_ នៅតែចាស់ ដូច្នេះអ្នកប្រើដែលមិនប្រើភាសាអង់គ្លេសនៅតែអានអត្ថបទដែលមានពាក្យពេចន៍ជឿជាក់ ប៉ុន្តែឥឡូវនេះខុស។

វាត្រូវបានដាក់ឱ្យប្រើប្រាស់ជាក់ស្តែង។ `oauthModal.googleOAuthWarning` ត្រូវបានសរសេរឡើងវិញ នៅពេលជំនួយការ login របស់ Antigravity ត្រូវបានដាក់បញ្ចូល (#5203)។ **39 ក្នុងចំណោម 43 locales** នៅតែមានអត្ថបទប្រាប់ប្រតិបត្តិករឱ្យ “ចម្លង URL ពេញលេញ ហើយបិទភ្ជាប់វានៅខាងក្រោម” — ជាលំហូរដែលមិនអាចបញ្ចប់បានសម្រាប់ provider នោះ។ វាមិនត្រូវបានកត់សម្គាល់រហូតដល់ #8463 ដោយសារ៖

- `sync-ui-keys` បំពេញបន្ថែមតែ keys ដែល **អវត្តមាន** ប៉ុណ្ណោះ មិនដែលបំពេញ keys ដែល **ចាស់** ទេ;

- `check-translation-drift` តាមដានឯកសារចម្លង documentation `docs/i18n/<locale>/**.md` —
  វាមិនដែលអាន `src/i18n/messages/*.json` ទេ។ កំពុងរារាំងក្នុង job `docs-sync-strict` ចាប់ពីការធ្វើសមកាលកម្មឡើងវិញនៅ

**យល់ដឹងពី diff មិនមែនពឹងផ្អែកលើ baseline ទេ។** វាប្រៀបធៀប `en.json` នៅ merge base ជាមួយ working tree; សម្រាប់គ្រប់ key ដែលតម្លៃភាសាអង់គ្លេសបានផ្លាស់ប្តូរ locale ណាមួយដែលនៅតែរក្សាការបកប្រែដែលមិនបានប៉ះពាល់ គឺចាស់។ វាចេតនា **បង្កកបំណុលដែលមានស្រាប់** — diff មិនអាចបង្ហាញថាការបកប្រែដែលមានមកយូរហើយនោះបានមកពីភាសាអង់គ្លេសចាស់មួយណាទេ ដូច្នេះ gate វាយតម្លៃតែអ្វីដែលការផ្លាស់ប្តូរបច្ចុប្បន្នប៉ះពាល់ប៉ុណ្ណោះ។ ជម្រើសផ្សេងទៀត (baseline hash សម្រាប់រាល់ key) នឹងចំណាយឯកសារដែលបានបង្កើតប្រហែល 600 KB ដែលធំជាង baseline ដែលមានស្រាប់ធំបំផុត 3 ដង ហើយបង្កឱ្យមានការផ្លាស់ប្តូររាល់ i18n PR។

## គោលការណ៍បញ្ជីអនុញ្ញាត

រាល់ច្រកត្រួតពិនិត្យដែលមិនអាចបរាជ័យដោយសារការបំពានដែលមានស្រាប់ ប្រើបញ្ជីអនុញ្ញាតដែលបានកំណត់ថេរ
(ឧ., `KNOWN_STALE_DOC_REFS`, `KNOWN_MISSING`, `KNOWN_RAW_SQL`)។ គោលការណ៍គឺ៖

**កែសម្រួលមូលហេតុដើម; ប្រើបញ្ជីអនុញ្ញាតតែក្នុងករណីដែលការបំពានមានស្រាប់ និង
មិនអាចកែសម្រួលបាននៅក្នុង PR ដូចគ្នា។**

នៅពេលបន្ថែមធាតុទៅបញ្ជីអនុញ្ញាត៖

1. រួមបញ្ចូលមតិយោបល់ដែលបញ្ជាក់ពីហេតុផល។
2. យោងទៅបញ្ហាតាមដាន (ឧ., `// #3498 — មុខងារដំណាក់កាលទី 2 មិនទាន់ត្រូវបានអនុវត្ត`)។
3. លុបធាតុនោះនៅក្នុង PR ដូចគ្នាដែលកែសម្រួលការបំពាន — ធាតុចាស់ដែលលែង
   ទប់ស្កាត់ការបំពានសកម្ម គឺជាកំហុសដោយខ្លួនឯង (ការអនុវត្តចំពោះធាតុចាស់ 6A.3 នឹង
   ធ្វើឱ្យច្រកត្រួតពិនិត្យបរាជ័យលើធាតុបញ្ជីអនុញ្ញាតដែលគ្មានការយោង នៅពេលវាត្រូវបានអនុវត្ត)។

**កុំ** បន្ថែមធាតុបញ្ជីអនុញ្ញាតដើម្បីឱ្យការធ្វើតេស្តឆ្លងកាត់លឿនជាងមុន។ ច្រកត្រួតពិនិត្យដែលជោគជ័យ ខណៈដែល

### នៅពេលច្រកត្រួតពិនិត្យបរាជ័យលើ PR របស់អ្នក

   នៅពេលកូដត្រឹមត្រូវ។

   ការផ្លាស់ប្តូររបស់អ្នកបានធ្វើឱ្យម៉ែត្រវាស់កាន់តែអាក្រក់។ កែសម្រួលបញ្ហាមូលដ្ឋាន ឬ (ក្នុងករណីកម្រ) ដំណើរការ
